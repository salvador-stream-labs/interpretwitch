cmake_minimum_required(VERSION 3.28)
project(awl-chat VERSION 0.8.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(AWL_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../.phase0/chat/deps" CACHE PATH "Prepared chat dependencies")
set(AWL_OBS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.phase0/research/obs-studio/obs-studio-32.2.2" CACHE PATH "OBS source")
set(AWL_OBS_APP /Applications/OBS.app CACHE PATH "Installed matching OBS.app")
if(WIN32)
  set(AWL_CT2_DEFAULT_LIBRARY "${AWL_DEPS}/bin/ctranslate2.dll")
else()
  set(AWL_CT2_DEFAULT_LIBRARY "${AWL_DEPS}/lib/libawl-ctranslate2.dylib")
endif()
set(AWL_CT2_LIBRARY "${AWL_CT2_DEFAULT_LIBRARY}" CACHE FILEPATH "CTranslate2 native library")
option(AWL_BUILD_PLUGIN "Build the OBS dock" ON)
option(AWL_OBS_SDK "Windows: use an OBS SDK CMake package instead of the installed OBS" OFF)
if(MSVC)
  add_compile_options(/utf-8 $<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>)
endif()
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network WebSockets)
find_package(Threads REQUIRED)
add_library(awl-ct2 SHARED IMPORTED)
set_target_properties(awl-ct2 PROPERTIES IMPORTED_LOCATION "${AWL_CT2_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${AWL_DEPS}/ct2/include")
if(WIN32)
  set(AWL_CT2_IMPLIB "${AWL_DEPS}/lib/ctranslate2.lib" CACHE FILEPATH "CTranslate2 Windows import library matching its DLL")
  set_target_properties(awl-ct2 PROPERTIES IMPORTED_IMPLIB "${AWL_CT2_IMPLIB}")
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ABSL_PROVIDER internal CACHE STRING "" FORCE)
add_subdirectory("${AWL_DEPS}/sentencepiece" sentencepiece EXCLUDE_FROM_ALL)
file(GLOB fasttext_sources "${AWL_DEPS}/fasttext/src/*.cc")
list(FILTER fasttext_sources EXCLUDE REGEX "/main.cc$")
add_library(awl-language STATIC ${fasttext_sources})
target_include_directories(awl-language PUBLIC "${AWL_DEPS}/fasttext/src")
target_link_libraries(awl-language PUBLIC Threads::Threads)
add_library(awl-chat-text STATIC src/ui_text.cpp)
target_include_directories(awl-chat-text PUBLIC src)
target_link_libraries(awl-chat-text PUBLIC Qt6::Core)
qt_add_resources(awl-chat-text chat_locale PREFIX "/awl-chat" BASE data FILES data/locale/ru-RU.json data/languages.json)
add_library(awl-chat-core STATIC src/chat_core.cpp src/model_profile.cpp src/diagnostics.cpp)
target_include_directories(awl-chat-core PUBLIC src)
target_link_libraries(awl-chat-core PUBLIC awl-chat-text)
# Inference and file access are helper-only targets. Neither is linked into OBS.
add_library(awl-chat-model STATIC src/model.cpp)
target_include_directories(awl-chat-model PRIVATE "${AWL_DEPS}/sentencepiece/src")
target_link_libraries(awl-chat-model PUBLIC awl-chat-core PRIVATE awl-ct2 sentencepiece-static awl-language Threads::Threads)
add_library(awl-chat-model-files STATIC src/model_files.cpp src/model_download.cpp)
target_include_directories(awl-chat-model-files PUBLIC src)
target_link_libraries(awl-chat-model-files PUBLIC awl-chat-core)
if(WIN32)
  # Optional NVIDIA cuBLAS pack for GPU mode: download, verify and inflate in the
  # helper. Only zlib's decompression sources are compiled.
  add_library(awl-zlib STATIC "${AWL_DEPS}/zlib/adler32.c" "${AWL_DEPS}/zlib/crc32.c"
    "${AWL_DEPS}/zlib/inffast.c" "${AWL_DEPS}/zlib/inflate.c" "${AWL_DEPS}/zlib/inftrees.c"
    "${AWL_DEPS}/zlib/zutil.c")
  target_include_directories(awl-zlib PUBLIC "${AWL_DEPS}/zlib")
  target_compile_definitions(awl-zlib PRIVATE _CRT_SECURE_NO_DEPRECATE _CRT_NONSTDC_NO_DEPRECATE)
  target_sources(awl-chat-model-files PRIVATE src/gpu_runtime.cpp)
  target_link_libraries(awl-chat-model-files PRIVATE winhttp awl-zlib)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(awl-chat-model-files PRIVATE CURL::libcurl)
  target_compile_options(awl-chat-model-files PRIVATE -Wno-deprecated-declarations)
endif()
add_library(awl-chat-supervisor STATIC src/worker.cpp)
target_link_libraries(awl-chat-supervisor PUBLIC awl-chat-core)
add_executable(awl-chat-worker src/helper.cpp)
target_link_libraries(awl-chat-worker PRIVATE awl-chat-model awl-chat-model-files Threads::Threads)
if(APPLE)
  set_target_properties(awl-chat-worker PROPERTIES BUILD_RPATH "${AWL_DEPS}/lib;${AWL_DEPS}/qt/lib"
    INSTALL_RPATH "@executable_path/../Frameworks;${AWL_OBS_APP}/Contents/Frameworks")
endif()
add_library(awl-chat-ui STATIC src/panel.cpp src/inline_chat.cpp)
qt_add_resources(awl-chat-ui chat_inline PREFIX "/awl-chat" BASE data FILES data/inline_chat.js)
target_link_libraries(awl-chat-ui PUBLIC awl-chat-supervisor Qt6::Widgets Qt6::Network Qt6::WebSockets)

if(AWL_BUILD_PLUGIN)
  if(APPLE)
    set(OBS_RELEASE_CANDIDATE 0)
    set(OBS_BETA 0)
    configure_file("${AWL_OBS_SOURCE}/libobs/obsconfig.h.in" generated/obsconfig.h)
    add_library(OBS::libobs SHARED IMPORTED)
    set_target_properties(OBS::libobs PROPERTIES IMPORTED_LOCATION "${AWL_OBS_APP}/Contents/Frameworks/libobs.framework/libobs"
      INTERFACE_INCLUDE_DIRECTORIES "${AWL_OBS_SOURCE}/libobs;${CMAKE_CURRENT_BINARY_DIR}/generated;${AWL_DEPS}/simde")
    add_library(OBS::frontend SHARED IMPORTED)
    set_target_properties(OBS::frontend PROPERTIES IMPORTED_LOCATION "${AWL_OBS_APP}/Contents/Frameworks/libobs-frontend-api.1.dylib"
      INTERFACE_INCLUDE_DIRECTORIES "${AWL_OBS_SOURCE}/frontend/api")
  elseif(WIN32 AND NOT AWL_OBS_SDK)
    # Link against the installed OBS through import libraries generated by
    # scripts/prepare-windows.py; headers come from the matching OBS source.
    set(OBS_RELEASE_CANDIDATE 0)
    set(OBS_BETA 0)
    configure_file("${AWL_OBS_SOURCE}/libobs/obsconfig.h.in" generated/obsconfig.h)
    add_library(OBS::libobs SHARED IMPORTED)
    set_target_properties(OBS::libobs PROPERTIES IMPORTED_IMPLIB "${AWL_DEPS}/lib/obs.lib"
      INTERFACE_INCLUDE_DIRECTORIES "${AWL_OBS_SOURCE}/libobs;${CMAKE_CURRENT_BINARY_DIR}/generated")
    add_library(OBS::frontend SHARED IMPORTED)
    set_target_properties(OBS::frontend PROPERTIES IMPORTED_IMPLIB "${AWL_DEPS}/lib/obs-frontend-api.lib"
      INTERFACE_INCLUDE_DIRECTORIES "${AWL_OBS_SOURCE}/frontend/api")
  else()
    find_package(libobs REQUIRED CONFIG)
    find_package(obs-frontend-api REQUIRED CONFIG)
    add_library(OBS::frontend ALIAS OBS::obs-frontend-api)
  endif()
  add_library(awl-chat MODULE src/plugin.cpp src/twitch_dock.cpp)
  add_dependencies(awl-chat awl-chat-worker)
  target_link_libraries(awl-chat PRIVATE awl-chat-ui OBS::libobs OBS::frontend)
  target_compile_definitions(awl-chat PRIVATE AWL_CHAT_VERSION="${PROJECT_VERSION}")
  set_target_properties(awl-chat PROPERTIES PREFIX "")
  if(APPLE)
    set_target_properties(awl-chat PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin SUFFIX ""
      BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/../Frameworks;@executable_path/../Frameworks"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Info.plist.in")
    target_link_options(awl-chat PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/scripts/exports.txt")
    install(TARGETS awl-chat LIBRARY DESTINATION .)
    install(TARGETS awl-chat-worker RUNTIME DESTINATION awl-chat.plugin/Contents/MacOS)
    install(DIRECTORY data/ DESTINATION awl-chat.plugin/Contents/Resources)
  elseif(WIN32)
    install(TARGETS awl-chat LIBRARY DESTINATION awl-chat/bin/64bit)
    install(TARGETS awl-chat-worker RUNTIME DESTINATION awl-chat/bin/64bit)
    install(DIRECTORY data/ DESTINATION awl-chat/data)
  endif()
endif()

# Review and fault-injection code is separate from the production target graph.
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
