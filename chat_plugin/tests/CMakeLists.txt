# Test fixtures and diagnostic tools are never installed.
# Preserve the documented build-directory commands when using this subdirectory.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
# OBS's Windows Qt SDK has no offscreen platform. Use minimal for headless tests
# and the native platform where layout depends on real font metrics.
if(WIN32)
  set(awl_headless_qpa minimal)
  set(awl_layout_qpa windows)
else()
  set(awl_headless_qpa offscreen)
  set(awl_layout_qpa offscreen)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME chat-locale-catalogue COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_locales.py")
add_executable(awl-chat-localisation-tests localisation_tests.cpp)
target_link_libraries(awl-chat-localisation-tests PRIVATE awl-chat-ui)
add_dependencies(awl-chat-localisation-tests awl-chat-worker)
add_test(NAME chat-localisation COMMAND awl-chat-localisation-tests $<TARGET_FILE:awl-chat-worker>)
set_tests_properties(chat-localisation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${awl_layout_qpa}" TIMEOUT 30)
if(AWL_BUILD_PLUGIN AND (APPLE OR WIN32))
  add_executable(awl-chat-load-test module_load.cpp)
  target_link_libraries(awl-chat-load-test PRIVATE OBS::libobs Qt6::Widgets Qt6::Network)
  if(APPLE)
    set_target_properties(awl-chat-load-test PROPERTIES BUILD_RPATH "${AWL_OBS_APP}/Contents/Frameworks;${AWL_DEPS}/qt/lib")
  endif()
endif()
